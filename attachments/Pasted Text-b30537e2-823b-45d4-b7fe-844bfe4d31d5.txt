```tsx
async function loadAudio(file?: File) {
  if (!file) return;

  if (audio?.url) URL.revokeObjectURL(audio.url);

  if (instrumental) {
    URL.revokeObjectURL(instrumental);
    setInstrumental(null);
  }

  const url = URL.createObjectURL(file);
  setAudioBusy(true);

  try {
    const context = getAudioContext();
    const buffer = await context.decodeAudioData(await file.arrayBuffer());
    const bpm = estimateBpm(buffer);

    setCurrentTime(0);
    setAudio({
      name: file.name,
      url,
      duration: buffer.duration,
      bpm,
      channels: buffer.numberOfChannels,
      sampleRate: buffer.sampleRate,
      buffer,
      waveform: createWaveformPoints(buffer),

    toast(`Audio loaded · BPM estimated ${bpm}`);
  } catch {
    URL.revokeObjectURL(url);
    toast("That audio format could not be read in this browser");
  } finally {
    setAudioBusy(false);
  }
}
```

`decodeAudioData()` asynchronously decodes complete `ArrayBuffer` audio data into an `AudioBuffer`, which matches this local upload flow. [3]

### 5. Update `removeVocals`

Replace the existing function:

```tsx
function removeVocals() {
  if (!audio) return;

  if (audio.channels < 2) {
    toast("Use a stereo file for center-vocal reduction");
    return;
  }

  setAudioBusy(true);

  window.setTimeout(() => {
    try {
      const ctx = getAudioContext();
      const out = ctx.createBuffer(
        2,
        audio.buffer.length,
        audio.buffer.sampleRate
      );

      const left = audio.buffer.getChannelData(0);
      const right = audio.buffer.getChannelData(1);
      const outputLeft = out.getChannelData(0);
      const outputRight = out.getChannelData(1);

      for (let index = 0; index < audio.buffer.length; index += 1) {
        const sample = (left[index] - right[index]) * 0.7;
        outputLeft[index] = sample;
        outputRight[index] = sample;
      }

      if (instrumental) URL.revokeObjectURL(instrumental);

      setInstrumental(URL.createObjectURL(encodeWav(out)));
      toast("Center-vocal-reduced WAV created");
    } catch {
      toast("Vocal reduction failed");
    } finally {
      setAudioBusy(false);
    }
  }, 40);
}
```

`AudioBuffer` exposes individual channel sample arrays through `getChannelData()`, which is the correct data source for this kind of PCM transformation. [4]
